let MESSAGES = [

  { id:'m2', side:'player', time:'21:04', status:'read',   kr:'그냥 집에 있어', zh:'就待在家裡' },

let platform = 'kakao';

/* ⚠️ 在這裡填入你的 Gemini API Key ⚠️ */
const GEMINI_API_KEY = '在這裡貼上你的API_Key';

const esc = s => String(s ?? '').replace(/[&<>"]/g,
  c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));

const statusText = s => ({ sent:'已傳送', unread:'未讀', read:'已讀' }[s] || '');

const nowTime = () => {
  const t = new Date();
  return String(t.getHours()).padStart(2,'0') + ':' + String(t.getMinutes()).padStart(2,'0');
};

/* ============================================================

function render() {
  const d = DATA[platform];
  document.getElementById('phone').className = 'phone ' + platform;

  /* 頂部欄 */
  document.getElementById('hdrSlot').innerHTML = `
    <header class="hdr">
      <div class="back">‹</div>
      <div class="avatar" style="background:${d.avatar}"></div>
      <div class="info">
        <div class="name">${esc(d.name)}</div>
        <div class="sub">${esc(d.sub)}</div>
      </div>
      <div class="tag">${platform === 'kakao' ? 'KAKAO' : 'IG'}</div>
    </header>`;

  /* 訊息 */
  const chat = document.getElementById('chat');
  let html = '';
  let prevSide = null;

  for (const m of MESSAGES) {
    const cont = prevSide === m.side;
    prevSide = m.side;

    let inner = '';

    if (m.kind === 'story_reply' && platform === 'ig') {
      inner += `<div class="ctx">回覆了你的限時動態</div>`;
    }

    if (m.kind === 'reel_share' || m.kind === 'post_share') {
      inner += `<div class="media">
        <div class="media-thumb"></div>
        <div class="media-cap">
          ${esc(m.mediaCaption || '')}
          ${m.mediaCaptionZh ? `<div class="zh">${esc(m.mediaCaptionZh)}</div>` : ''}
        </div>
      </div>`;
    }

    if (m.kr) inner += `<div class="kr">${esc(m.kr)}</div>`;
    if (m.zh) inner += `<div class="zh">${esc(m.zh)}</div>`;
    if (!inner) inner = '<div class="kr">·</div>';

    const meta = m.side === 'player'
      ? `<span>${esc(m.time)}</span>` +
        (m.status ? `<span class="st-${m.status}">${statusText(m.status)}</span>` : '')
      : `<span>${esc(m.time)}</span>`;

    html += `
      <div class="row ${m.side}${cont ? ' cont' : ''}">
        ${m.side === 'npc'
          ? `<div class="av" style="background:${d.avatar}"></div>`
          : ''}
        <div class="col">
          ${m.side === 'npc' && !cont
            ? `<div class="sender">${esc(d.name)}</div>`
            : ''}
          <div class="line">
            <div class="bubble">${inner}</div>
            <div class="meta">${meta}</div>
          </div>
          ${m.reactions && m.reactions.length
            ? `<div class="reactions">${m.reactions.join('')}</div>`
            : ''}
        </div>
      </div>`;
  }

  if (isTyping) {
    html += `<div class="typing">對方正在輸入…</div>`;
  }

  chat.innerHTML = html;
  chat.scrollTop = chat.scrollHeight;
}

/* ============================================================
   5. 呼叫 Gemini API
   ============================================================ */
async function callAI(userMessage) {
  if (!GEMINI_API_KEY || GEMINI_API_KEY === '在這裡貼上你的API_Key') {
    alert('請先在程式碼中填入 Gemini API Key！');
    return null;
  }

  const prompt = `你是一個韓語戀愛模擬角色，名字叫朴智旻。你正在和用戶用聊天軟體聊天。請根據用戶的話，用自然的韓語回覆，並且提供繁體中文翻譯。格式必須是：韓語原文|繁體中文翻譯。不要有多餘的解釋。用戶說：${userMessage}`;

  const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${GEMINI_API_KEY}`;

  const body = {
    contents: [{
      parts: [{ text: prompt }]
    }]
  };

  try {
    const res = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body)
    });
    const data = await res.json();
    if (data.candidates && data.candidates[0] && data.candidates[0].content) {
      return data.candidates[0].content.parts[0].text;
    } else {
      console.error('API 回傳異常:', data);
      return '미안, 지금 답장 못 해 😢|抱歉，現在沒辦法回你 😢';
    }
  } catch (err) {
    console.error('API 呼叫失敗：', err);
    return '미안, 지금 답장 못 해 😢|抱歉，現在沒辦法回你 😢';
  }
}

/* ============================================================
   6. 發送訊息 + AI 回覆
   ============================================================ */
async function send() {
  const inp = document.getElementById('inp');
  const v = inp.value.trim();
  if (!v) return;

  const isKr = /[\uAC00-\uD7AF]/.test(v);
  const playerMsg = {
    id: 'u' + Date.now(),
    side: 'player',
    time: nowTime(),
    status: 'sent'
  };
  playerMsg[isKr ? 'kr' : 'zh'] = v;

  MESSAGES.push(playerMsg);
  inp.value = '';
  render();

  isTyping = true;
  render();

  const aiReplyRaw = await callAI(v);
  if (aiReplyRaw === null) {
    isTyping = false;
    render();
    return;
  }

  let kr = aiReplyRaw;
  let zh = '';
  if (aiReplyRaw.includes('|')) {
    const parts = aiReplyRaw.split('|');
    kr = parts[0].trim();
    zh = parts[1] ? parts[1].trim() : '';
  }

  const lastPlayer = [...MESSAGES].reverse().find(m => m.side === 'player');
  if (lastPlayer) lastPlayer.status = 'read';

  MESSAGES.push({
    id: 'ai' + Date.now(),
    side: 'npc',
    time: nowTime(),
    kr: kr,
    zh: zh
  });

  isTyping = false;
  render();
}
